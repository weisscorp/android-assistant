cmake_minimum_required(VERSION 3.22)
project(localbrain LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
add_subdirectory(../../../../vendor/llama.cpp llama)
add_library(localbrain SHARED brain.cpp)
target_link_libraries(localbrain PRIVATE llama-common llama android log)
target_include_directories(localbrain PRIVATE ../../../../vendor/llama.cpp/common ../../../../vendor/llama.cpp/vendor)
# Catalog validation belongs to the actual JNI implementation as well as upstream llama.cpp.
set(LLAMA_PINNED_REVISION "d81235049384534c167caea52b85a694f6103d14")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/brain.cpp")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/brain.cpp" JNI_SOURCE_SHA)
file(SHA256 "${CMAKE_CURRENT_LIST_FILE}" JNI_CMAKE_SHA)
string(SHA256 JNI_FINGERPRINT "${LLAMA_PINNED_REVISION}:${JNI_SOURCE_SHA}:${JNI_CMAKE_SHA}")
string(SUBSTRING "${JNI_FINGERPRINT}" 0 16 JNI_FINGERPRINT_SHORT)
target_compile_definitions(localbrain PRIVATE ASSISTANT_LLAMA_REVISION="${LLAMA_PINNED_REVISION}"
    ASSISTANT_BINDING_REVISION="${LLAMA_PINNED_REVISION}:jni-${JNI_FINGERPRINT_SHORT}")
target_compile_options(localbrain PRIVATE -O3)
target_link_options(localbrain PRIVATE "-Wl,-z,max-page-size=16384")

# Small CPU echo/denoise component; pinned upstream sources, no codec/network dependency.
set(SPEEX_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/speexdsp)
add_library(voicedsp SHARED echo.cpp
    ${SPEEX_ROOT}/libspeexdsp/mdf.c ${SPEEX_ROOT}/libspeexdsp/preprocess.c
    ${SPEEX_ROOT}/libspeexdsp/fftwrap.c ${SPEEX_ROOT}/libspeexdsp/filterbank.c
    ${SPEEX_ROOT}/libspeexdsp/smallft.c)
target_include_directories(voicedsp PRIVATE ${SPEEX_ROOT}/include ${SPEEX_ROOT}/libspeexdsp speex-config)
target_compile_definitions(voicedsp PRIVATE FLOATING_POINT USE_SMALLFT EXPORT=)
target_compile_options(voicedsp PRIVATE -O3)
target_link_libraries(voicedsp PRIVATE m log)
target_link_options(voicedsp PRIVATE "-Wl,-z,max-page-size=16384")
